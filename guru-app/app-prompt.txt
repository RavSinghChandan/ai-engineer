# graph/state.py
from typing import TypedDict, List, Optional, Dict, Any

class LifeEvent(TypedDict):
    event_type: str
    domain: str
    raw_data: Dict[str, Any]
    timestamp: str
    user_id: str

class DomainState(TypedDict):
    domain: str
    current_score: float
    insights: List[str]
    recommendations: List[str]
    alerts: List[str]
    trend: str  # "improving" | "stable" | "declining"

class AntaryamiState(TypedDict):
    user_id: str
    current_event: LifeEvent
    semantic_memory: Dict[str, Any]
    episodic_memory: List[Dict]
    behavioral_patterns: Dict[str, Any]
    user_persona: Dict[str, Any]
    unified_life_state: Dict[str, Any]
    active_agents: List[str]
    emotional_tone: str
    intervention_severity: str
    finance_state: Optional[DomainState]
    health_state: Optional[DomainState]
    spiritual_state: Optional[DomainState]
    career_state: Optional[DomainState]
    cross_domain_insights: List[str]
    life_score: float
    unified_recommendation: str
    actions_to_execute: List[Dict]
    reflection_summary: str
    scores_updated: bool
    orchestrator_decision: Dict[str, Any]
    cycle_complete: bool
    error: Optional[str]

## BASE AGENT CONTRACT (Plug-and-Play)

# graph/agents/base_agent.py
from abc import ABC, abstractmethod
from graph.state import AntaryamiState, DomainState

class BaseLifeAgent(ABC):
    """
    Plug-and-play contract for all domain agents.

    To add a new domain in V2:
    1. Create new_agent.py
    2. Inherit from BaseLifeAgent
    3. Implement the 3 abstract methods
    4. Register in graph/builder.py (3 lines)
    Nothing else changes. Zero breaking changes.
    """

    @property
    @abstractmethod
    def domain_name(self) -> str:
        pass

    @abstractmethod
    def analyze(self, state: AntaryamiState) -> DomainState:
        pass

    @abstractmethod
    def get_system_prompt(self) -> str:
        pass

    def should_activate(self, state: AntaryamiState) -> bool:
        return self.domain_name in state["active_agents"]

    def run(self, state: AntaryamiState) -> AntaryamiState:
        if not self.should_activate(state):
            return state
        domain_result = self.analyze(state)
        state[f"{self.domain_name}_state"] = domain_result
        return state

## GRAPH BUILDER

# graph/builder.py
from langgraph.graph import StateGraph, END
from langgraph.checkpoint.memory import MemorySaver
from graph.state import AntaryamiState

def build_antaryami_graph():
    graph = StateGraph(AntaryamiState)

    graph.add_node("normalize", normalize_event)
    graph.add_node("retrieve_memory", retrieve_memory)
    graph.add_node("assemble_state", assemble_state)
    graph.add_node("orchestrate", orchestrate)
    graph.add_node("finance_agent", FinanceAgent().run)
    graph.add_node("health_agent", HealthAgent().run)
    graph.add_node("spiritual_agent", SpiritualAgent().run)
    graph.add_node("career_agent", CareerAgent().run)
    graph.add_node("synthesize", synthesize_cross_domain)
    graph.add_node("execute_actions", execute_actions)
    graph.add_node("reflect", run_reflection)
    graph.add_node("write_memory", write_back_memory)

    graph.set_entry_point("normalize")
    graph.add_edge("normalize", "retrieve_memory")
    graph.add_edge("retrieve_memory", "assemble_state")
    graph.add_edge("assemble_state", "orchestrate")

    graph.add_conditional_edges(
        "orchestrate",
        lambda s: s["active_agents"],
        {
            "finance": "finance_agent",
            "health": "health_agent",
            "spiritual": "spiritual_agent",
            "career": "career_agent",
        }
    )

    for agent in ["finance_agent","health_agent","spiritual_agent","career_agent"]:
        graph.add_edge(agent, "synthesize")

    graph.add_edge("synthesize", "execute_actions")
    graph.add_edge("execute_actions", "reflect")
    graph.add_edge("reflect", "write_memory")
    graph.add_edge("write_memory", END)

    return graph.compile(checkpointer=MemorySaver())
